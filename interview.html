function renderAnalysis(analysis) { let statusText = "AI分析が完了しました。"; if (analysis.status === "ready") { statusText = "EA仕様を作成できる状態です。"; } if ( analysis.status === "needs_clarification" ) { statusText = "EA仕様を確定するため、追加確認が必要です。"; } if ( analysis.status === "parse_error" ) { statusText = "AIの回答をJSONとして解析できませんでした。"; } let html = `

AI分析結果

${escapeHTML(statusText)}
`; if (analysis.summary) { html += `

手法の概要

${escapeHTML( analysis.summary )}
`; } html += renderStrategy( analysis.strategy ); /* ========================= AIからの追加質問＋回答欄 ========================= */ if ( Array.isArray( analysis.additionalQuestions ) && analysis.additionalQuestions.length > 0 ) { html += `

AIからの追加質問

EA仕様を完成させるため、 以下の質問に回答してください。

`; analysis.additionalQuestions .forEach( (question, index) => { const questionText = typeof question === "string" ? question : JSON.stringify(question); html += `
${index + 1}. ${escapeHTML( questionText )}
`; } ); html += `
`; } html += renderList( "確認が必要な矛盾", analysis.contradictions, "warning" ); html += renderList( "AIの推奨設定候補", analysis.reasonableDefaults, "default" ); html += renderList( "リスク上の注意", analysis.riskWarnings, "warning" ); if (analysis.raw) { html += `
AIの生データを確認 
${escapeHTML(
          analysis.raw
        )}
`; } html += `
`; resultEl.innerHTML = html; /* 現在の質問を一時保存 */ window.currentAdditionalQuestions = Array.isArray( analysis.additionalQuestions ) ? analysis.additionalQuestions : []; } /* ========================= 追加質問の回答を保存して AIへ再送信 ========================= */ function submitClarifications() { const questions = window.currentAdditionalQuestions || []; if (!questions.length) { alert( "追加質問が見つかりません。" ); return; } const clarificationAnswers = {}; for ( let index = 0; index < questions.length; index++ ) { const input = document.getElementById( `clarification-${index}` ); const value = input ? input.value.trim() : ""; if (!value) { alert( `${index + 1}番目の質問に回答してください。` ); if (input) { input.focus(); } return; } const question = typeof questions[index] === "string" ? questions[index] : JSON.stringify( questions[index] ); clarificationAnswers[ question ] = value; } /* 既存ヒアリング回答を取得 */ let savedAnswers = {}; try { const raw = localStorage.getItem( "eaInterviewAnswers" ); if (raw) { const parsed = JSON.parse(raw); if ( parsed && typeof parsed === "object" && !Array.isArray(parsed) ) { savedAnswers = parsed; } } } catch (error) { console.error( "ヒアリング回答の読込エラー:", error ); } /* 追加回答をヒアリング回答へ追加 */ savedAnswers.aiClarifications = { ...( savedAnswers.aiClarifications || {} ), ...clarificationAnswers }; localStorage.setItem( "eaInterviewAnswers", JSON.stringify( savedAnswers ) ); /* AI再分析 */ analyzeEA(); }
